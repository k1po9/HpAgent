import { render, screen, waitFor } from '@testing-library/react';
import { afterEach, beforeEach, expect, it, vi } from 'vitest';
import { RunResources } from './RunResources';
import { runApi, useRunInspector } from '../../store/runInspector';
import { createTraceStore } from '../trace/traceStore';
import { HpCommandError } from '../../api/types';
beforeEach(() => { useRunInspector.getState().reset(); useRunInspector.getState().select('A'); });
afterEach(() => vi.restoreAllMocks());
it('R1: refreshes published output when a visible Run reaches terminal state', async () => {
  vi.spyOn(runApi, 'listRunResources').mockResolvedValue({count:0,next:null,candidates:[]});
  vi.spyOn(runApi, 'listRunFileApprovals').mockResolvedValue({approvals:[]});
  const outputs=vi.spyOn(runApi, 'listRunPublishedFiles').mockResolvedValueOnce({files:[]}).mockResolvedValue({files:[{file_id:'result',name:'result.txt',sha256:'safe'}]});
  const view=render(<RunResources runId="A" terminal={false} />);
  await screen.findByText('暂无已发布输出。');
  view.rerender(<RunResources runId="A" terminal />);
  await waitFor(() => expect(outputs).toHaveBeenCalledTimes(2));
  expect(screen.getByRole('link',{name:'result.txt'})).toBeInTheDocument();
});
it('R2: permission denial invalidates cached and in-flight Model Input bodies', async () => {
  let resolveLate!: (value:unknown)=>void;
  const full=(id:string)=>({visibility:'full_safe',model_input:{snapshot_id:id,provider_request_body:{messages:['sensitive fixture']}}});
  const api={getModelInput:vi.fn().mockResolvedValueOnce(full('first')).mockImplementationOnce(()=>new Promise(resolve=>{resolveLate=resolve;})).mockRejectedValueOnce(new HpCommandError(403,{code:'model_input_unavailable',message:'denied',request_id:null,retryable:false,details:{}}))};
  const store=createTraceStore(api as never);
  store.getState().selectRun('A');
  await store.getState().loadModelInput('first');
  const late=store.getState().loadModelInput('late');
  await store.getState().loadModelInput('denied');
  resolveLate(full('late'));
  await late;
  expect(Object.values(store.getState().modelInputs).filter(value=>value.detail?.model_input.provider_request_body)).toHaveLength(0);
});
